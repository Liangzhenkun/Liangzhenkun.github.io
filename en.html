// Learning basic React Hooks usage
import React, { useState, useEffect } from 'react';

function DataFetchingComponent() {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    // Simulate data fetching
    const fetchData = async () => {
      try {
        const response = await fetch('https://api.example.com/data');
        const result = await response.json();
        setData(result);
        setLoading(false);
      } catch (error) {
        console.error('Error fetching data:', error);
        setLoading(false);
      }
    };
    
    fetchData();
  }, []); // Empty dependency array means it only runs on component mount
  
  return (
    <div>
      {loading ? (
        <p>Loading data...</p>
      ) : (
        <ul>
          {data.map(item => (
            <li key={item.id}>{item.name}</li>
          ))}
        </ul>
      )}
    </div>
  );
}

Notes from my frontend learning journey, just starting to learn React. I've experienced the advantages of declarative UI and how functional components simplify code structure. As a Ph.D. student, learning programming in my spare time is both a hobby and a way to enhance my research capabilities.

Research Data Processing Script

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Some data visualization experiments
def analyze_research_data(file_path):
    """Analyze experimental data and generate charts"""
    # Read data
    df = pd.read_csv(file_path)
    
    # Data cleaning
    df = df.dropna()
    df = df[(df['reaction_time'] > 100) & (df['reaction_time'] < 3000)]
    
    # Basic statistics
    stats = df.groupby('condition').agg({
        'reaction_time': ['mean', 'std', 'count']
    })
    
    # Visualization
    plt.figure(figsize=(10, 6))
    sns.boxplot(x='condition', y='reaction_time', data=df)
    plt.title('Reaction time distribution under experimental conditions')
    plt.tight_layout()
    plt.savefig('reaction_time_by_condition.png')
    
    return stats

# Usage example
if __name__ == "__main__":
    results = analyze_research_data("experiment_data.csv")
    print(results)

Python is my得力助手 for processing large amounts of data in research. This script was created for behavioral experiment data analysis, combining pandas and seaborn for data processing and visualization.
